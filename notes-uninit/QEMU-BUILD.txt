Plugin-enabled qemu-aarch64 (qemu-aarch64-static in ~/.local/bin has no TCG plugin support):
  mkdir /tmp/uninitsem_qemu && cd /tmp/uninitsem_qemu
  curl -sSLO https://download.qemu.org/qemu-10.2.1.tar.xz && tar xf qemu-10.2.1.tar.xz
  # glib headers without root: apt download libglib2.0-dev libglib2.0-dev-bin libgio-2.0-dev libpcre2-dev
  #   libffi-dev zlib1g-dev libmount-dev libselinux-dev libsepol-dev libblkid-dev libsysprof-capture-4-dev
  # dpkg-deb -x each into sysroot/; point dangling sysroot/usr/lib/x86_64-linux-gnu/*.so links at /usr/lib/x86_64-linux-gnu/*
  cd qemu-10.2.1 && mkdir build && cd build
  PKG_CONFIG_PATH=$S/usr/lib/x86_64-linux-gnu/pkgconfig:$S/usr/share/pkgconfig PKG_CONFIG_SYSROOT_DIR=$S \
    ../configure --target-list=aarch64-linux-user --enable-plugins --disable-docs --disable-werror --disable-tools
  ninja -j12 qemu-aarch64          # ~35 s on 32 cores
